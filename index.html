async function goToPart4() { showScreen('part4'); const container = document.getElementById('cards-stack'); container.innerHTML = ''; cardsSwipedCount = 0; document.getElementById('btn-to-part5').style.display = 'none'; let allCards = []; let ownedIds = []; try { const { data: cData } = await _supabase.from('cards').select('*'); const { data: iData } = await _supabase.from('user_inventory').select('card_id'); if(cData) allCards = cData; if(iData) ownedIds = iData.map(i => i.card_id); } catch(e) {} if (allCards.length === 0) { allCards = [{ id: 1, name: 'Carta Prova', rarity: 'Comune', image_url: 'https://picsum.photos/200/280' }]; } // Raggruppa le carte per rarità const cardsByRarity = { 'Comune': allCards.filter(c => c.rarity === 'Comune'), 'Rara': allCards.filter(c => c.rarity === 'Rara'), 'Epica': allCards.filter(c => c.rarity === 'Epica'), 'Leggendaria': allCards.filter(c => c.rarity === 'Leggendaria') }; // Funzione per estrarre una carta in base alle percentuali function getRandomCardByRarity() { const rand = Math.random() * 100; // Numero casuale tra 0 e 100 let targetRarity = 'Comune'; if (rand < 5) { targetRarity = 'Leggendaria'; // 5% chance } else if (rand < 15) { targetRarity = 'Epica'; // 10% chance } else if (rand < 40) { targetRarity = 'Rara'; // 25% chance } else { targetRarity = 'Comune'; // 60% chance } // Se non ci sono carte per quella rarità, prende una lista di ripiego let pool = cardsByRarity[targetRarity]; if (!pool || pool.length === 0) pool = allCards; return pool[Math.floor(Math.random() * pool.length)]; } drawnCards = []; for (let i = 0; i < 4; i++) { const randomCard = getRandomCardByRarity(); const isDuplicate = ownedIds.includes(randomCard.id); drawnCards.push({ ...randomCard, isDuplicate }); } drawnCards.forEach((card, index) => { const wrapper = document.createElement('div'); wrapper.className = 'card-wrapper'; wrapper.style.zIndex = 4 - index; let dupHTML = card.isDuplicate ? `
DOPPIONE (+10 Polvere)
` : ''; wrapper.innerHTML = `
${card.name}
${card.rarity}
${dupHTML}
`; let startY = 0; wrapper.addEventListener('touchstart', e => { startY = e.touches[0].clientY; }); wrapper.addEventListener('touchend', e => { if (startY - e.changedTouches[0].clientY > 30) performSwipe(wrapper); }); wrapper.addEventListener('mousedown', e => { startY = e.clientY; }); wrapper.addEventListener('mouseup', e => { if (startY - e.clientY > 30) performSwipe(wrapper); }); container.appendChild(wrapper); }); }
